Fixpoint var_t_op (x: tname) (t:typ) : variance :=
  let var_t_bound_op :=
    fun (tbnd:t_bound) =>
      let (b, t) := tbnd in
        var_meet_op (negateVar (var_b_op x b)) (var_t_op x t)
  in
  let var_t_bounds_op :=
      fun (tbnds:t_bounds) =>
        let vars := List.map var_t_bound_op tbnds in
          var_meet_list_op vars
  in
  match t with
    | t_fvar y => if x == y then covar else bivar
    | t_bivar _ _ => bivar
    | t_ext tbnds N =>
        var_meet_op (var_t_bounds x tbnds) (var_n x N)
  end

  let (m, _) := md in m.
